   19: def _clean_owner_env_value(raw: object, fallback: str = "") -> str:
   20:     value = str(raw or "").strip().strip('"').strip("'")
   21: 
   22:     if not value:
   23:         return fallback
   24: 
   25:     value = value.replace("\r", "\n")
   26:     parts = [part.strip() for part in value.split("\n") if part.strip()]
   27: 
   28:     # Handles bad Render paste:
   29:     # OWNER_API_TOKEN
   30:     # wolf-owner-local-token
   31:     if len(parts) >= 2 and parts[0].upper().replace(" ", "_") in {"OWNER_API_TOKEN", "ADMIN_PASSWORD"}:
   32:         value = parts[-1]
   33: 
   34:     # Handles bad Render paste:
   35:     # OWNER_API_TOKEN=wolf-owner-local-token
   36:     if "=" in value:
   37:         left, right = value.split("=", 1)
   38:         if left.strip().upper().replace(" ", "_") in {"OWNER_API_TOKEN", "ADMIN_PASSWORD"}:
   39:             value = right.strip()
   40: 
   41:     return value.strip().strip('"').strip("'") or fallback

  627: def require_owner() -> tuple[bool, Any]:
  628:     auth = request.headers.get("Authorization", "")
  629:     token = ""
  630:     if auth.lower().startswith("bearer "):
  631:         token = auth.split(" ", 1)[1].strip()
  632: 
  633:     if secrets.compare_digest(token, owner_token()):
  634:         return True, None
  635: 
  636:     return False, (jsonify({"ok": False, "error": "Unauthorized owner request."}), 401)
